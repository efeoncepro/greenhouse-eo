import fs from 'node:fs'
import { planSurfacePiece, type SurfaceIntent } from '@/lib/brand-surfaces'
const [intentPath, outDir] = process.argv.slice(2)
const piece = planSurfacePiece(JSON.parse(fs.readFileSync(intentPath, 'utf8')) as SurfaceIntent, { artifactId: 'plate' })
fs.mkdirSync(outDir, { recursive: true })
for (const a of piece.assets as Array<Record<string, unknown>>) {
  const keys = Object.keys(a)
  const body = (a.svg ?? a.content ?? a.data) as string | undefined
  console.log(a.ref, a.kind, keys.join(','), typeof body === 'string' ? body.length : '')
  if (typeof body === 'string' && String(a.ref).includes('stage')) fs.writeFileSync(`${outDir}/stage.svg`, body)
}
