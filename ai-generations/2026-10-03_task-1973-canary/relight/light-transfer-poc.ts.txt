import sharp from 'sharp'
import { maskFromLayers } from '../layers'
import { maskStats } from '../mask'
import { cutElement, pasteElement } from '../move'
import { encodeRgbaPng, loadRgba } from '../raw'
import { measureZones } from '../recompose'

const R = 'ai-generations/2026-10-03_task-1973-canary'
const W = 1536, H = 1024

// Desenfoque gaussiano aproximado: 3 pasadas de caja separables (radio de cada caja ≈ sigma).
const boxPass = (src: Float32Array, r: number, horizontal: boolean): Float32Array => {
  const out = new Float32Array(src.length)
  const len = horizontal ? W : H, lines = horizontal ? H : W
  for (let l = 0; l < lines; l++) {
    let acc = 0
    const idx = (k: number) => (horizontal ? l * W + k : k * W + l)
    for (let k = -r; k <= r; k++) acc += src[idx(Math.min(len - 1, Math.max(0, k)))]
    for (let k = 0; k < len; k++) {
      out[idx(k)] = acc / (2 * r + 1)
      acc += src[idx(Math.min(len - 1, k + r + 1))] - src[idx(Math.max(0, k - r))]
    }
  }
  return out
}

const blurPlane = async (plane: Float32Array, sigma: number): Promise<Float32Array> => {
  let p = plane
  const r = Math.max(1, Math.round(sigma))
  for (let k = 0; k < 3; k++) p = boxPass(boxPass(p, r, true), r, false)
  return p
}

const main = async () => {
  const L = `${R}/layers/7ba99ac215de/layers.json`
  const source = await loadRgba(`${R}/base.png`)
  const element = await maskFromLayers(L, ['mug'], { width: source.width, height: source.height })
  const box = maskStats(element).bbox!
  const target = await loadRgba(`${R}/destino-pared-verde.png`)
  const { composed, placed } = await pasteElement(target, cutElement(source, element, box), { left: 207, top: 369, width: 261, height: 225 })
  const sunburst = await loadRgba(`${R}/relight/place/7106910349f1/inpaint/0fde0812b198/candidate-1.png`)
  const ic = await loadRgba(await sharp(`${R}/relight/place/7106910349f1/inpaint/0b2aa657339e/candidate-1-raw.png`).resize(W, H, { fit: 'fill' }).png().toBuffer())

  const lum = (d: Uint8Array, i: number) => 0.299 * d[i * 4] + 0.587 * d[i * 4 + 1] + 0.114 * d[i * 4 + 2]
  const n = W * H
  const M = new Float32Array(n), LA = new Float32Array(n), LB = new Float32Array(n)
  const chA = [0, 1, 2].map(() => new Float32Array(n)), chB = [0, 1, 2].map(() => new Float32Array(n))
  for (let i = 0; i < n; i++) {
    const m = placed.data[i] / 255
    M[i] = m; LA[i] = lum(composed.data, i) * m; LB[i] = lum(ic.data, i) * m
    for (let c = 0; c < 3; c++) { chA[c][i] = composed.data[i * 4 + c] * m; chB[c][i] = ic.data[i * 4 + c] * m }
  }
  for (const sigma of [6, 14]) {
    const [Mb, LAb, LBb] = await Promise.all([blurPlane(M, sigma), blurPlane(LA, sigma), blurPlane(LB, sigma)])
    const sigmaC = sigma * 2.5
    const [Mc, ...rest] = await Promise.all([blurPlane(M, sigmaC), ...chA.map(p => blurPlane(p, sigmaC)), ...chB.map(p => blurPlane(p, sigmaC)), blurPlane(LA, sigmaC), blurPlane(LB, sigmaC)])
    const cA = rest.slice(0, 3), cB = rest.slice(3, 6), lAc = rest[6], lBc = rest[7]
    // Parte del resultado de Sunburst (sombra de contacto) y repone los píxeles ORIGINALES del objeto.
    const out = new Uint8Array(sunburst.data)
    for (let i = 0; i < n; i++) {
      const m = placed.data[i] / 255
      if (!m) continue
      const g = Math.min(2.5, Math.max(0.4, (LBb[i] / (Mb[i] || 1)) / Math.max(1, LAb[i] / (Mb[i] || 1))))
      for (let c = 0; c < 3; c++) {
        const tint = Math.min(1.25, Math.max(0.8, (cB[c][i] / Math.max(1, lBc[i])) / Math.max(1e-3, cA[c][i] / Math.max(1, lAc[i]))))
        const v = Math.min(255, composed.data[i * 4 + c] * g * tint)
        out[i * 4 + c] = Math.round(v * m + sunburst.data[i * 4 + c] * (1 - m))
      }
    }
    const final = { ...sunburst, data: out }
    const z = measureZones(target, final, (await import('../mask')).dilate(placed, 60))
    console.log(`sigma ${sigma}: fuera de taza+60px delta max ${z.protected.maxDelta}`)
    await sharp(await encodeRgbaPng(final)).toFile(`${R}/relight/light-transfer-s${sigma}.png`)
  }
}
void main()
